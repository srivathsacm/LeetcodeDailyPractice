class Solution {
    public int largestMagicSquare(int[][] grid) {
        int m = grid.length, n = grid[0].length;

        int[][] row = new int[m][n + 1];
        int[][] col = new int[m + 1][n];
        int[][] d1 = new int[m + 1][n + 1];
        int[][] d2 = new int[m + 1][n + 2];

        for (int i = 0; i < m; i++) {
            for (int j = 0; j < n; j++) {
                row[i][j + 1] = row[i][j] + grid[i][j];
                col[i + 1][j] = col[i][j] + grid[i][j];
                d1[i + 1][j + 1] = d1[i][j] + grid[i][j];
                d2[i + 1][j] = d2[i][j + 1] + grid[i][j];
            }
        }

        for (int k = Math.min(m, n); k > 1; k--) {
            for (int i = 0; i + k <= m; i++) {
                for (int j = 0; j + k <= n; j++) {
                    int sum = row[i][j + k] - row[i][j];
                    boolean ok = true;

                    for (int t = 0; t < k && ok; t++) {
                        if (row[i + t][j + k] - row[i + t][j] != sum) ok = false;
                        if (col[i + k][j + t] - col[i][j + t] != sum) ok = false;
                    }

                    if (!ok) continue;

                    if (d1[i + k][j + k] - d1[i][j] != sum) continue;
                    if (d2[i + k][j] - d2[i][j + k] != sum) continue;

                    return k;
                }
            }
        }
        return 1;
    }
}
