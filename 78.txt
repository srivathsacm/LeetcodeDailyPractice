import java.util.*;

class Solution {
    public long maxSumTrionic(int[] nums) {
        int n = nums.length;
        long NEG = -(1L << 60);

        long[] inc1 = new long[n];
        long[] dec = new long[n];
        long[] inc2 = new long[n];

        Arrays.fill(inc1, NEG);
        Arrays.fill(dec, NEG);
        Arrays.fill(inc2, NEG);

        for (int i = 1; i < n; i++) {
            if (nums[i] > nums[i - 1]) {
                inc1[i] = nums[i - 1] + nums[i];
                if (inc1[i - 1] > NEG)
                    inc1[i] = Math.max(inc1[i], inc1[i - 1] + nums[i]);
            }
        }

        for (int i = 2; i < n; i++) {
            if (nums[i] < nums[i - 1]) {
                if (inc1[i - 1] > NEG)
                    dec[i] = inc1[i - 1] + nums[i];

                if (dec[i - 1] > NEG)
                    dec[i] = Math.max(dec[i], dec[i - 1] + nums[i]);
            }
        }

        long ans = NEG;

        for (int i = 3; i < n; i++) {
            if (nums[i] > nums[i - 1]) {
                if (dec[i - 1] > NEG)
                    inc2[i] = dec[i - 1] + nums[i];

                if (inc2[i - 1] > NEG)
                    inc2[i] = Math.max(inc2[i], inc2[i - 1] + nums[i]);
            }
            ans = Math.max(ans, inc2[i]);
        }

        return ans;
    }
}
